/* =========================================================
   TABELA: CLIENTES
   ========================================================= */

CREATE TABLE Clientes (
    ID INT PRIMARY KEY,
    Nome NVARCHAR(100),
    Telefone VARCHAR(20),
    Endereco NVARCHAR(200),
    CPF VARCHAR(14),
    Nivel INT,
    Senha VARCHAR(255),
    Usuario NVARCHAR(50)
);


/* =========================================================
   TABELA: ANIMAIS
   ========================================================= */

CREATE TABLE Animais (
    ID INT PRIMARY KEY,
    Nome NVARCHAR(80),
    Raca NVARCHAR(60),
    Sexo CHAR(1),
    Enfermidade NVARCHAR(150),
    Descricao NVARCHAR(500),
    Especie NVARCHAR(30),
    fk_Clientes_ID INT
);


/* =========================================================
   TABELA: VETERINARIOS
   ========================================================= */

CREATE TABLE Veterinarios (
    ID INT PRIMARY KEY,
    Nome NVARCHAR(100),
    Telefone VARCHAR(20),
    Nivel INT,
    Usuario NVARCHAR(50),
    Senha VARCHAR(255)
);


/* =========================================================
   TABELA: CONSULTAS
   ========================================================= */

CREATE TABLE Consultas (
    Diagnostico NVARCHAR(500),
    Preco MONEY,
    ID INT PRIMARY KEY,
    DataHora DATETIME,
    fk_Animais_ID INT,
    fk_Veterinarios_ID INT
);


/* =========================================================
   TABELA: SERVICOS
   ========================================================= */

CREATE TABLE Servicos (
    ID INT PRIMARY KEY,
    Tipo INT
);


/* =========================================================
   TABELA: MEDICAMENTOS
   ========================================================= */

CREATE TABLE Medicamentos (
    ID INT PRIMARY KEY,
    Nome NVARCHAR(100),
    Descricao NVARCHAR(500),
    Quantidade INT,
    Preco MONEY,
    fk_Estoque_Financeiro_ID INT
);


/* =========================================================
   TABELA: PRODUTOS
   ========================================================= */

CREATE TABLE Produtos (
    ID INT PRIMARY KEY,
    Nome NVARCHAR(100),
    Descricao VARCHAR(500),
    Quantidade INT,
    Preco MONEY,
    fk_Estoque_Financeiro_ID INT
);


/* =========================================================
   TABELA: FORNECEDORES
   ========================================================= */

CREATE TABLE Fornecedores (
    ID INT PRIMARY KEY,
    Nome NVARCHAR(100),
    CPF_CNPJ VARCHAR(18),
    Telefone VARCHAR(20),
    Descricao NVARCHAR(300)
);


/* =========================================================
   TABELA: ESTOQUE_FINANCEIRO
   ========================================================= */

CREATE TABLE Estoque_Financeiro (
    ID INT PRIMARY KEY,
    Entrada MONEY,
    Saida MONEY,
    Saldo MONEY,
    fk_Fornecedores_ID INT
);


/* =========================================================
   TABELA: SERVICO_CONSULTA_REALIZA
   ========================================================= */

CREATE TABLE Servico_Consulta_Realiza (
    Preco_Cobrado MONEY,
    Quantidade INT,
    Observacoes NVARCHAR(300),
    fk_Consultas_ID INT,
    fk_Servicos_ID INT
);


/* =========================================================
   TABELA: MEDICAMENTO_CONSULTA_UTILIZA
   ========================================================= */

CREATE TABLE Medicamento_Consulta_Utiliza (
    fk_Medicamentos_ID INT,
    fk_Consultas_ID INT
);
